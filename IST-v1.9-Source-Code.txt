@echo off

:: 32-Bit platform detection
if "%PROCESSOR_ARCHITECTURE%"=="x86" (

        echo Please restart IST in normal mode.
        echo ERROR-CODE: 0x32

        exit /b
        GOTO :EOF
    ) else (
        echo SYSERROR: IST cannot run on a 32-Bit OS.
        echo So, IST is not supported on your machine.
        echo ERROR-CODE: 0x32
        pause
        exit /b
        GOTO :EOF
    )

:: No PowerShell detection

:: Versions

    FOR /F "tokens=*" %%i IN ('where ist.bat') DO SET istDir=%%i

    set "istPgf=%PROGRAMFILES%\IST"

    set "admStat=0"
    set "admLab=[USER]"

set "line="

set /a "padding=(cols - 35) / 2"
set "spaces="
for /l %%i in (1,1,%padding%) do set "spaces=!spaces! "

:: YOLO through DeadZone
set "YOLO=0"

        set "YOLO=1"
    ) else if "!yoloMode!"=="YOLOFLAG" (
        if "%~2"=="--yolo" set "YOLO=1"
    ) else if "!yoloMode!"=="YOLONO" (
        set "YOLO=0"
    )
)

:: Log Maintenance: Prevent infinite growth
if exist "%istPgf%\log-files" (

)

:: Sad smiley
set "sSM=:("


:: Other startup tasks 

:: Title Variable

:: Title

:: Config Files
    :: IP Address

    for /f "tokens=*" %%i in ('type "%istPgf%\bin\ipType.ist"') do set "ipType=%%i"

        set "ipType=public"
    )
    if "!ipType!"=="private" (
        set "ipType=private"
    )

    set "savedIpAddress=0"
)
    :: Wi-Fi name
if exist "%istPgf%\bin\wifiName.ist" (
    set "savedWifiName=Exists"
) else (
    set "savedWifiName=0"
)

:: Configuration for theme loader
if exist "%istPgf%\bin\themeManager.ist" (
    for /f "tokens=*" %%i in ('type "%istPgf%\bin\themeManager.ist"') do set "themeCfg=%%i"

    if "!themeCfg!"=="Ocean "     color 17
    if "!themeCfg!"=="WinPWSH "   color 1F
    if "!themeCfg!"=="Blood "     color 04

    if "!themeCfg!"=="Volcano "   color 4E
    if "!themeCfg!"=="Ubuntu "    color 0D
    if "!themeCfg!"=="WarnArea "  color 06
    if "!themeCfg!"=="OLDmacOS "  color 8F
    if "!themeCfg!"=="Default "   color 07

    echo ERROR-CODE: 0x2
    GOTO :EOF
)

if not exist "%ISTPGF%\bin" mkdir "%ISTPGF%\bin"
if not exist "%ISTPGF%\log-files" mkdir "%ISTPGF%\log-files"
echo [ OK ] Integrity restored.
GOTO FLAGS
GOTO :EOF

:RADM_CHECK
:: Admin Restart disabled?
if "!RADM!"=="RADMNO" (
    GOTO :EOF
)

:: Admin Restart Automatic?
if "!RADM!"=="RADMAUTO" (
    echo [ NOTICE ] command %1 requires administrator rights.
    echo But, IST will restart as administrator automatically.
    timeout /t 1 >nul 2>&1
    echo Restarting as administrator...
    pause
    powershell -NoProfile -Command "Start-Process cmd -ArgumentList @('/k', 'ist %1') -Verb RunAs"
    set "restarted=1"
    exit /b
    GOTO :EOF
)

if "!RADM!"=="RADMASK" (
    echo [ NOTICE ] command %1 requires administrator rights.
    set /p "radmChoice=Do you want to restart as administrator? (y/n): "
    if /i "!radmChoice!"=="y" (
        powershell -NoProfile -Command "Start-Process cmd -ArgumentList @('/k', 'ist %1') -Verb RunAs"
        set "restarted=1"
        exit /b
        GOTO :EOF
        
    )
    if /i "!radmChoice!"=="n" (
        echo Aborted, please restart the command as administrator manually.
        exit /b
        GOTO :EOF
    )
    echo Invalid selection. Please use y/n.
    echo ERROR-CODE: 0x102
    GOTO :EOF
    GOTO :EOF
)
GOTO :EOF

:IP
echo Showing your IP address is a serious business! Do not show this during a stream or someone that you don't want unauthorized access to your IP address. 
set /p "ynIp=Are you sure you want to continue? (y/n): "
if "%ynIp%"=="y" (
    if "%2"=="--private" (
        :: This shows only your local IP info
        echo Your IP Information:
        ipconfig | findstr "IPv4"
        GOTO :EOF
    )
    if "%2"=="--public" (
        :: This shows only your public IP info
        echo Your IP Information:
        powershell -NoProfile -Command "(Invoke-RestMethod -Uri 'https://api.ipify.org')"
        GOTO :EOF
    )
    :: This shows your local and public IP info
    echo Your IP Information:
    ipconfig | findstr "IPv4"
    echo.
    echo Your Public IP:
    powershell -NoProfile -Command "(Invoke-RestMethod -Uri 'https://api.ipify.org')"
    GOTO :EOF
)
if "%ynIp%"=="n" (
    echo Aborted, security risk forbidden.
    GOTO :EOF
)
GOTO :EOF

    echo Pinging %target% continuously... Press CTRL+C to stop
    ping %target% -t
) else (
    echo Pinging %target%...
    ping %target%
)

:SPEEDTESTDRIVER
where winget >nul 2>nul
if %errorlevel% EQU 0 (
    echo Installing Speedtest CLI via WinGet...
    winget install --id Ookla.Speedtest.CLI -e --silent --accept-source-agreements --accept-package-agreements
) else (
    echo [!] WinGet not found. Falling back to manual download...
    powershell -NoProfile -Command "Invoke-WebRequest -Uri 'https://install.speedtest.net/app/cli/ookla-speedtest-1.2.0-win64.zip' -OutFile '%TEMP%\speedtest.zip'"
    powershell -NoProfile -Command "Expand-Archive -Path '%TEMP%\speedtest.zip' -DestinationPath '%istPgf%\external-scripts' -Force"
    echo [ OK ] Speedtest CLI manually installed to %istPgf%\external-scripts.
)
GOTO :EOF

:DRIVERWINGET
where winget >nul 2>nul
if %errorlevel% EQU 0 (
    echo [ NOTICE ] Winget is already installed on the system.
    GOTO :EOF
) else (
    echo [!] WinGet not found. Attempting to install WinGet...
    powershell -NoProfile -Command ^
    "$progressPreference='silentlyContinue';" ^
    "Write-Host 'Downloading WinGet...';" ^
    "Invoke-WebRequest -Uri 'https://aka.ms/getwinget' -OutFile '%TEMP%\winget.msixbundle';" ^
    "Write-Host 'Downloading VCLibs dependency...';" ^
    "Invoke-WebRequest -Uri 'https://aka.ms/Microsoft.VCLibs.x64.14.00.Desktop.appx' -OutFile '%TEMP%\vclibs.appx';" ^
    "Write-Host 'Downloading UIXaml dependency...';" ^
    "Invoke-WebRequest -Uri 'https://github.com/microsoft/microsoft-ui-xaml/releases/download/v2.8.6/Microsoft.UI.Xaml.2.8.x64.appx' -OutFile '%TEMP%\uixaml.appx';" ^
    "Write-Host 'Installing...';" ^
    "Add-AppxPackage '%TEMP%\vclibs.appx';" ^
    "Add-AppxPackage '%TEMP%\uixaml.appx';" ^
    "Add-AppxPackage '%TEMP%\winget.msixbundle'"
    echo [ OK ] WinGet installed. Please restart IST.
    GOTO :EOF
)

GOTO :EOF
