    echo [ OK ] You are up to date (%version%)
) else (
    echo Update available! - %latest_version%
    set /p "YNUPD=Do you want to update to %latest_version%? (y/n) "
    if "!YNUPD!"=="y" (
        echo Downloading...
        set "zipURL=https://github.com/Stefi9/IST/releases/latest/download/IST.zip"
        set "destination=%USERPROFILE%\Downloads\update.zip"
        powershell -NoProfile -Command "Invoke-WebRequest -Uri '!zipURL!' -OutFile '!destination!'"
        echo Extracting...
        powershell -NoProfile -Command "Expand-Archive -Path '!destination!' -DestinationPath '%ISTPGF%' -Force"
        echo [ OK ] IST has been updated to !latest_version!.
    ) else (
        echo Aborted, IST stays at version %version%
    )
)
